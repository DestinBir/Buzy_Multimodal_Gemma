\begin{frame}{The Problem: Document Chaos in African Organizations}
  \begin{columns}[T]
    \column{0.6\textwidth}
    Across Africa, SMEs, cooperatives, and NGOs manage critical decisions using documents scattered across paper, PDFs, images, and audio recordings.
    \vspace{0.5em}
    \begin{itemize}
      \item \alert{No centralized knowledge} — contracts, invoices, meeting notes live in silos
      \item \alert{Manual analysis} takes hours or days
      \item \alert{Institutional knowledge walks away} when staff leave
      \item \alert{Language barriers} — French, English, Lingala, Swahili documents mixed
    \end{itemize}
    \vspace{0.5em}
    \textbf{Real questions organizations ask:}
    \begin{quote}
      ``Which supplier is most reliable for our cooperative?''
    \end{quote}
    \begin{quote}
      ``Why did our Q2 expenses exceed the budget?''
    \end{quote}

    \column{0.4\textwidth}
    \centering
    \includegraphics[width=\textwidth,height=0.7\textheight,keepaspectratio]{business_knowledge_graph.png}
  \end{columns}
\end{frame}

\begin{frame}{Why This Matters for DRC \& Africa}
  \begin{columns}[T]
    \column{0.6\textwidth}
    \begin{itemize}
      \item \textbf{80\% of African SMEs} operate without formal document management systems
      \item \textbf{Manual contract review} costs small businesses \$500+ per contract in lost productivity
      \item \textbf{Meeting decisions} are lost when only recorded as audio — no transcription, no search
      \item \textbf{Multilingual reality} — French administration, local languages for operations
    \end{itemize}
    \vspace{0.5em}
    \textbf{Buzy AI bridges the gap:} no expensive ERP, no cloud subscription — runs on a laptop, works offline, speaks French and English.

    \column{0.4\textwidth}
    \centering
    \includegraphics[width=\textwidth,keepaspectratio]{architecture.png}
  \end{columns}
\end{frame}

\begin{frame}{Solution: Buzy AI}
  \textbf{Buzy AI} is a multimodal document intelligence system that transforms raw business documents into structured, explainable decisions.
  \vspace{0.5em}
  \begin{columns}[T]
    \column{0.6\textwidth}
    \begin{enumerate}
      \item \textbf{Ingest} PDFs, DOCX, images (receipts, dashboards), and audio recordings
      \item \textbf{Extract} structured facts using Gemma 4 Vision-Language
      \item \textbf{Build} a searchable knowledge base
      \item \textbf{Retrieve} relevant evidence for any business question
      \item \textbf{Generate} structured recommendations with source citations
    \end{enumerate}
    \vspace{0.5em}
    \textbf{AI for Local Impact:} Designed for African organizations with unreliable internet, mixed document formats, and multilingual needs.

    \column{0.4\textwidth}
    \centering
    \includegraphics[width=\textwidth,keepaspectratio]{business_decision_pipeline.png}
  \end{columns}
\end{frame}

\begin{frame}{Live Scenario: Supplier Risk Analysis}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \textbf{Context:} A Congolese procurement cooperative manages 6 suppliers

    \textbf{Question:} Which supplier represents the highest operational risk?
    \vspace{0.5em}
    \begin{itemize}
      \item \alert{Evidence:} 18-day payment delay average, delivery instability flagged in 2 meetings, contract expiring in 27 days
      \item \alert{Reasoning:} Repeated delays + sole-source dependency on critical project
      \item \alert{Confidence:} \textcolor{gdggreen}{High (0.82)}
    \end{itemize}

    \column{0.45\textwidth}
    \textbf{Impact:} 2-week project delay risk; unfavorable renewal terms

    \textbf{Actions:}
    \begin{enumerate}
      \item Begin renewal negotiations immediately
      \item Qualify a secondary supplier
      \item Escalate payment approvals
    \end{enumerate}
    \vspace{0.5em}
    \alert{All recommendations cite exact source documents}
  \end{columns}
\end{frame}
